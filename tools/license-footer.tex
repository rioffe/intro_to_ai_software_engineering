% license-footer.tex -- a LaTeX preamble that adds the open-educational material
% branding ("CC BY 4.0 - (c) 2026 Robert Ioffe") as a per-page footer WITHOUT
% disturbing the book class's running headers.
%
% Wired in by tools/build-book-localtoc.sh as a second --include-in-header
% (unless LICENSE=0).
%
% What each page gets:
%   * body pages: the book class's "headings" running header, unchanged --
%     UPPERCASE slanted marks (section on odd, chapter on even), page number in
%     the outer corner, no rule -- PLUS a centered "CC BY 4.0 ..." footer line.
%   * chapter openers / Contents / front matter: no running header (as before),
%     with the license flush-left and the page number flush-right in the foot.
%
% IMPLEMENTATION NOTES:
%  * We rebuild "headings" via fancyhdr rather than adding a foot to the class
%    style, because you cannot append to \ps@headings cleanly.  \chaptermark /
%    \sectionmark are restored to the class's UPPERCASE, prefix-free form (the
%    manuscript numbers its own headings; LaTeX's secnumdepth is negative).
%  * Odd/even \fancyhead[..] codes are used only in the HEADER (where the class
%    already splits by side); the footer is a single centered \fancyfoot[C] that
%    lands on every page, avoiding fancyhdr's flaky LE/RO footer behaviour.
%  * Re-assert \pagestyle{fancy} around \frontmatter / \mainmatter: pandoc's
%    book output runs \frontmatter \maketitle \tableofcontents \mainmatter, and
%    that sequence otherwise leaves the class default in effect for the body.
%  * @-free macro names, so no \makeatletter (pandoc's include-in-header runs
%    with a normal catcode for '@').  To change the wording, edit \LicenseFooter.

\usepackage{fancyhdr}

\newcommand{\LicenseFooter}{%
      \small CC BY\ 4.0\ \textbullet\ \copyright\ 2026\ \ \textbf{Robert\ Ioffe}}

% Marks exactly as the book class's \ps@headings would set them.
\renewcommand{\chaptermark}[1]{\markboth{\MakeUppercase{#1}}{}}
\renewcommand{\sectionmark}[1]{\markright{\MakeUppercase{#1}}}

% Running-header pages: the class "headings" layout + a centered license foot.
\newcommand{\LicenseMainStyle}{%
      \fancyhf{}%
      \fancyhead[LE]{\thepage}\fancyhead[RE]{\slshape\leftmark}%
      \fancyhead[LO]{\slshape\rightmark}\fancyhead[RO]{\thepage}%
      \fancyfoot[C]{\LicenseFooter}%
      \renewcommand{\headrulewidth}{0pt}%
      \renewcommand{\footrulewidth}{0pt}}

% No-running-header pages (chapter openers, Contents, front matter): the page
% number moves down beside the license.
\newcommand{\LicensePlainStyle}{%
      \fancyhf{}%
      \fancyfoot[C]{\LicenseFooter \hfill \thepage}%
      \renewcommand{\headrulewidth}{0pt}%
      \renewcommand{\footrulewidth}{0pt}}

% The cover page (page 1) is full-bleed artwork.  "empty" is redefined just
% below to still carry the licence line, so the cover needs a style of its own
% that really is bare; build-book-localtoc.sh points \CoverPageStyle at it.
\fancypagestyle{coverbare}{%
      \fancyhf{}%
      \renewcommand{\headrulewidth}{0pt}%
      \renewcommand{\footrulewidth}{0pt}}
\renewcommand{\CoverPageStyle}{coverbare}

\pagestyle{fancy}
\fancypagestyle{fancy}{\LicenseMainStyle}
\fancypagestyle{plain}{\LicensePlainStyle}
\fancypagestyle{empty}{\LicensePlainStyle}

\AtBeginDocument{\pagestyle{fancy}}
\let\LicenseOrigFrontmatter\frontmatter
\renewcommand{\frontmatter}{\LicenseOrigFrontmatter\pagestyle{fancy}}
\let\LicenseOrigMainmatter\mainmatter
\renewcommand{\mainmatter}{\LicenseOrigMainmatter\pagestyle{fancy}}
